# Crontab of the deploy user on the VPS (#20, F15): crontab -u deploy deploy/crontab.txt
# The jobs run from current/, so a deploy or a rollback changes what runs without touching this file.
# The server's time zone is Europe/Zurich (timedatectl, docs/operations.md), so these times are Zurich.
# No warm-up cron on purpose (the 00:05 clear plus Neon's wake-ups would cost more than they save).
# cron mails a job's output to MAILTO; a good night prints nothing.
MAILTO=__OPS_MAIL__
#
# 00:05: empty the HTTP cache after the midnight expiry.
5 0 * * * /bin/sh /var/www/deine-eventlocation.ch/current/bin/cron-midnight
# Every 5 minutes, offset by 2 from the clear above and serialised with it by a lock (#19 n2):
# empty the cache only when it is over its size or file limit.
2-59/5 * * * * /bin/sh /var/www/deine-eventlocation.ch/current/bin/cron-cache-guard
# 03:15: inquiry purge, complainant erasure (D36), search-miss purge (D46), CRM stages and renewal tasks, stuck-inquiry report, stats ingest, pool prune, sweeps.
15 3 * * * /bin/sh /var/www/deine-eventlocation.ch/current/bin/cron-daily
# 04:30 (F15b): the media originals to Object Storage with restic, encrypted; an unchanged night copies nothing.
30 4 * * * /bin/sh /var/www/deine-eventlocation.ch/current/bin/cron-media-backup
